(()=>{
  'use strict';
  const nativeFetch=window.fetch.bind(window);
  const base='https://raw.githubusercontent.com/OtavioBoullon/latente-brasa-site/main/public/poupai-v4-full/';
  const load=async(name)=>{
    const r=await nativeFetch(base+name+'?cb='+Date.now(),{cache:'no-store'});
    if(!r.ok)throw new Error('Falha ao carregar '+name);
    const js=await r.text();
    const s=document.createElement('script');s.textContent=js;document.documentElement.appendChild(s);
  };
  (async()=>{
    try{
      await load('integration-v213.txt');
      await load('mobile-render-fix-v214.txt');
      await load('mobile-pdf-fast-v216.txt');
      await load('paywall-v212.txt');
    }catch(err){
      console.error('Poupai loader:',err);
      const t=document.querySelector('#toast');if(t){t.textContent='Não foi possível carregar a análise real. Atualize a página.';t.classList.add('show')}
    }
  })();
})();
