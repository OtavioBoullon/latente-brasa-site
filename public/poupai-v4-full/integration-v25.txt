(()=>{
  'use strict';
  const nativeFetch=window.fetch.bind(window);
  const src='https://raw.githubusercontent.com/OtavioBoullon/latente-brasa-site/main/public/poupai-v4-full/integration-v210.txt?cb='+Date.now();
  nativeFetch(src,{cache:'no-store'})

})();
