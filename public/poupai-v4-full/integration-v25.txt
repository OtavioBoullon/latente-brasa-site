      const typedCep=cleanCep($('#cep')?.value);
      if(typedCep)extraction.cep=typedCep;
      if(extraction.cep&&$('#cep'))$('#cep').value=extraction.cep;

      phase(30,'Validando a leitura…','Checando consistência, descontos, combos e custo efetivo da conta.','#c2');
      const preflight=await post('/api/run-pipeline',{extraction,marketResult:{},engineConfig:{}},20000);
      renderExtracted(extraction,preflight);
      if(preflight.decisionReason!=='INSUFFICIENT_MARKET_DATA'){
        finishLoader();renderPipeline(preflight,extraction);return;
      }

      const cep=cleanCep(extraction.cep||$('#cep')?.value);
      if(!cep){finishLoader();renderManualBlock(extraction,'Precisamos do CEP da instalação para pesquisar ofertas reais. Digite o CEP e clique em “Analisar agora” novamente.');return}

      let marketResult=marketPayload.result||{};

      const number=String($('#houseNumber')?.value||'').trim();
      if(number&&marketResult.offers?.length){

        if(supported.length){

            const checked=await post('/api/check-availability',{cep,number,providers:supported,marketResult},90000);
            if(checked.updatedMarket)marketResult=checked.updatedMarket;
          }catch(err){console.warn('Poupai checker:',err);}
        }
      }else{$('#c4')?.classList.add('done')}

      phase(90,'Calculando a recomendação…','Aplicando economia em 12/24 meses, qualidade e regras de segurança.','#c5');

    }
  }
